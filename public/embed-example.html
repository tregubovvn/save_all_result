Примеры кода интеграции

1. Встраивание через iframe 

<!-- Только ИД студента -->
<iframe 
  src="index.html?mode=embed&fields=studentId"
  allow="camera"
></iframe>

<!-- Все поля -->
<iframe 
  src="index.html?mode=embed&fields=studentId,activity,grade,comment"
></iframe>

2. Получение событий через postMessage 

window.addEventListener('message', function(event) {
  if (event.data.type === 'studentRecordSaved') {
    console.log('Запись сохранена:', event.data.record);
    // event.data.record содержит:
    // { studentId, activity, grade, comment, dateTime }
  }
  
  if (event.data.type === 'studentRecordError') {
    console.error('Ошибка:', event.data.message);
  }
});

3. Программное управление 

// Получить iframe
const iframe = document.getElementById('student-tracker-frame');

// Установить ИД студента
iframe.contentWindow.App.setStudentId('STU12345');

// Получить все записи
const records = iframe.contentWindow.App.getRecords();

// Очистить записи
iframe.contentWindow.App.clearRecords();

// Изменить конфигурацию
iframe.contentWindow.App.setConfig({
  fields: ['studentId', 'grade'],
  onSuccess: function(record) {
    console.log('Сохранено:', record);
  }
});

4. Доступные поля 

// Поля для параметра fields:
studentId  - ИД студента (с QR-сканированием)
activity   - Название активности
grade      - Оценка (0-5)
comment    - Комментарий

// Примеры комбинаций:
fields=studentId                          // Только ИД
fields=studentId,grade                    // ИД и оценка
fields=studentId,activity,grade           // ИД, активность, оценка
fields=studentId,activity,grade,comment   // Все поля
