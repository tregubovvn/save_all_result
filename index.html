Конфигурация nginx (Lua)

# nginx.conf с Lua модулем для записи в CSV
# Требуется: openresty или nginx + lua-nginx-module

server {
    listen 443 ssl http2;
    server_name your-domain.com;

    ssl_certificate     /etc/ssl/certs/server.crt;
    ssl_certificate_key /etc/ssl/private/server.key;

    # API endpoint - запись в CSV через Lua
    location /api/submit {
        content_by_lua_block {
            local csv_file = "/var/www/student-tracker/data/records.csv"

            -- Читаем тело запроса
            ngx.req.read_body()
            local body = ngx.req.get_body_data()

            -- Проверяем токен
            local token = ngx.req.get_headers()["x-request-token"]
            if not token then
                ngx.status = 401
                ngx.say('{"success":false,"message":"No token"}')
                return
            end

            -- Дешифруем данные (XOR + Base64)
            local data = ngx.decode_base64(body)
            -- XOR дешифрование выполняется на стороне Lua

            -- Проверяем существование файла
            local f = io.open(csv_file, "r")
            if not f then
                -- Создаём новый файл с заголовком
                f = io.open(csv_file, "w")
                f:write("Student ID,Activity,Grade,Comment,Date/Time,File URL\n")
                f:close()
            else
                f:close()
            end

            -- Добавляем запись
            f = io.open(csv_file, "a")
            f:write(data .. "\n")
            f:close()

            ngx.header["Content-Type"] = "application/json"
            ngx.header["Access-Control-Allow-Origin"] = "*"
            ngx.say('{"success":true,"message":"Запись сохранена"}')
        }

        # CORS preflight
        if ($request_method = 'OPTIONS') {
            add_header 'Access-Control-Allow-Origin' '*';
            add_header 'Access-Control-Allow-Methods' 'POST, OPTIONS';
            add_header 'Access-Control-Allow-Headers' 'Content-Type, X-Request-Token';
            return 204;
        }
    }

    # Статические файлы
    location / {
        root /var/www/student-tracker;
        try_files $uri $uri/ /index.html;
    }
}
